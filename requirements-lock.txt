# Fichier de verrouillage pour un build de release REPRODUCTIBLE
# (constat d'audit, 2026-07-28) - complement de requirements.txt (qui garde
# volontairement une plage ouverte ">=2.7" pour les contributeurs qui
# lancent TempoFacture depuis le code source), PAS un remplacement.
#
# requirements.txt seul ne garantit rien : deux builds de dist/TempoFacture.exe
# a des dates differentes peuvent embarquer des versions differentes de
# `fpdf2` (et de sa propre chaine de dependances : defusedxml, fonttools,
# Pillow) sans que cela soit trace nulle part. Les versions ci-dessous sont
# celles reellement utilisees dans l'environnement de developpement au moment
# de cet audit (verifiees via `pip show fpdf2` / `pip freeze`), figees
# explicitement plutot que laissees a la resolution de dependances du jour.
#
# A REGENERER et committer AVANT chaque nouvelle release, dans
# l'environnement de build reellement utilise pour PyInstaller :
#
#   python -m pip install --upgrade -r requirements.txt
#   python -m pip freeze | findstr /i "fpdf2 defusedxml fonttools pillow" > requirements-lock.txt
#
# Pour un build reproductible a partir de ce fichier :
#
#   python -m pip install -r requirements-lock.txt
#   python -m PyInstaller TempoFacture.spec
#
# PLANCHER DE SECURITE - Pillow >= 12.3.0 (CVE-2026-25990, CVE-2026-40192).
# TempoFacture ne tire Pillow qu'INDIRECTEMENT, par fpdf2, et n'embarque
# aucune image : l'exposition reelle est nulle. Mais GuideExpress, PdfAtelier
# et PhotoTri imposent ce plancher dans leur requirements.txt, et ce fichier
# est le controle de release de CE projet : le laisser epingler une version
# en dessous revenait a dire silencieusement le contraire des trois autres
# (constat C8 de l'audit du 2026-08-26 - le verrou disait aussi fpdf2 2.8.7
# quand l'environnement de developpement portait deja 2.8.8, exactement la
# derive que ce fichier existe pour empecher).
fpdf2==2.8.8
defusedxml==0.7.1
fonttools==4.63.0
pillow==12.3.0
